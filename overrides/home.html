team.py 
from anycode import AnyCode, AgentConfig, TaskSpec, TeamConfig

engine = AnyCode(config={"max_concurrency": 3})

team = engine.create_team("crew", TeamConfig(
    name="crew",
    shared_memory=True,
    agents=[
        AgentConfig(name="planner", provider="anthropic", model=MODEL),
        AgentConfig(name="builder", provider="anthropic", model=MODEL),
        AgentConfig(name="reviewer", provider="anthropic", model=MODEL),
    ],
))

result = await engine.run_tasks(team, [
    TaskSpec(title="Plan",   assignee="planner"),
    TaskSpec(title="Build",  assignee="builder",  depends_on=["Plan"]),
    TaskSpec(title="Review", assignee="reviewer", depends_on=["Build"]),
])  # runs in dependency-ordered waves
